cmake_minimum_required(VERSION 3.0.2)
project(uav_audio)

find_package(catkin REQUIRED COMPONENTS roscpp roslib message_generation)
find_package(Threads REQUIRED)
add_service_files(FILES PlayAudio.srv Say.srv)
generate_messages()
catkin_package(CATKIN_DEPENDS roscpp roslib message_runtime)

add_executable(audio_node src/audio_node.cpp)
set_target_properties(audio_node PROPERTIES CXX_STANDARD 14 CXX_STANDARD_REQUIRED YES)
target_compile_options(audio_node PRIVATE -Wall -Wextra -Wpedantic)
target_include_directories(audio_node PRIVATE ${catkin_INCLUDE_DIRS})
target_link_libraries(audio_node ${catkin_LIBRARIES} ${CMAKE_THREAD_LIBS_INIT})
add_dependencies(audio_node ${${PROJECT_NAME}_EXPORTED_TARGETS} ${catkin_EXPORTED_TARGETS})

set(SHERPA_ONNX_ROOT "" CACHE PATH "sherpa-onnx v1.12.27 shared SDK (optional)")
if(SHERPA_ONNX_ROOT)
  set(SHERPA_ONNX_INCLUDE_DIR "${SHERPA_ONNX_ROOT}/include")
  set(SHERPA_ONNX_LIBRARY_DIR "${SHERPA_ONNX_ROOT}/lib")
  set(SHERPA_ONNX_C_API "${SHERPA_ONNX_LIBRARY_DIR}/libsherpa-onnx-c-api.so")
  if(NOT EXISTS "${SHERPA_ONNX_INCLUDE_DIR}/sherpa-onnx/c-api/c-api.h"
      OR NOT EXISTS "${SHERPA_ONNX_C_API}")
    message(FATAL_ERROR "SHERPA_ONNX_ROOT must point to a sherpa-onnx shared SDK")
  endif()
  target_compile_definitions(audio_node PRIVATE UAV_AUDIO_WITH_TTS)
  target_include_directories(audio_node PRIVATE ${SHERPA_ONNX_INCLUDE_DIR})
  target_link_libraries(audio_node "${SHERPA_ONNX_C_API}")
  set_target_properties(audio_node PROPERTIES INSTALL_RPATH "${SHERPA_ONNX_LIBRARY_DIR}")
endif()

install(TARGETS audio_node RUNTIME DESTINATION ${CATKIN_PACKAGE_BIN_DESTINATION})
install(DIRECTORY audio launch DESTINATION ${CATKIN_PACKAGE_SHARE_DESTINATION})
